# ============================================================
# DATABASE
# ============================================================

# MySQL database driver for Python.
#
# Django's MySQL database backend needs a database driver to
# communicate with the actual MySQL server.
#
# Simple architecture:
#
#     Django
#        |
#        v
#     mysqlclient
#        |
#        v
#     MySQL
#
# Our project currently uses MySQL as its database.
#
# Without this package, Django may fail with an error such as:
#
#     No module named 'MySQLdb'
#
# In simple words:
# mysqlclient = Bridge between Django and MySQL.
mysqlclient==2.2.8


# PostgreSQL database driver for Python.
#
# This is the PostgreSQL equivalent of mysqlclient.
#
# Architecture:
#
#     Django
#        |
#        v
#     psycopg2
#        |
#        v
#     PostgreSQL
#
# The current project uses MySQL, so this package is not required
# for the current local MySQL setup.
#
# It is kept here because the project also documents PostgreSQL
# as an alternative production database.
#
# In simple words:
# psycopg2-binary = Bridge between Django and PostgreSQL.
psycopg2-binary==2.9.12


# Converts a DATABASE_URL connection string into the dictionary
# format expected by Django's DATABASES setting.
#
# Instead of writing:
#
#     ENGINE
#     NAME
#     USER
#     PASSWORD
#     HOST
#     PORT
#
# separately in settings.py, we can use one value in .env:
#
#     DATABASE_URL=mysql://root:@127.0.0.1:3307/hms_db
#
# Then settings.py can use:
#
#     dj_database_url.parse(DATABASE_URL)
#
# This makes switching between development and production
# databases much easier.
#
# In simple words:
# dj-database-url = Converts DATABASE_URL into Django DB settings.
dj-database-url==3.1.2


# ============================================================
# ENVIRONMENT VARIABLES / CONFIGURATION
# ============================================================

# Reads configuration values from environment variables or a
# local .env file.
#
# This is used for values such as:
#
#     SECRET_KEY
#     DEBUG
#     ALLOWED_HOSTS
#     DATABASE_URL
#     EMAIL_HOST
#     EMAIL_HOST_USER
#     EMAIL_HOST_PASSWORD
#
# Example .env:
#
#     SECRET_KEY=your-secret-key
#     DEBUG=True
#     DATABASE_URL=mysql://root:@127.0.0.1:3307/hms_db
#
# Then settings.py can use:
#
#     from decouple import config
#     SECRET_KEY = config("SECRET_KEY")
#
# This prevents sensitive information such as passwords and
# secret keys from being hardcoded in the source code.
#
# The .env file should normally be included in .gitignore.
#
# In simple words:
# python-decouple = Reads configuration/secrets from .env.
python-decouple==3.8


# ============================================================
# IMAGE PROCESSING / IMAGE VALIDATION
# ============================================================

# Pillow is the Python Imaging Library used for working with
# image files.
#
# This project allows users to upload profile pictures for:
#
#     - Patients
#     - Doctors
#     - Receptionists
#
# Pillow can:
#
#     - Open images
#     - Verify images
#     - Read image information
#     - Resize images
#     - Convert image formats
#     - Process image files
#
# The project uses Pillow to verify that an uploaded file is
# actually a valid image and not simply a file renamed with an
# image extension.
#
# Example:
#
#     from PIL import Image
#
#     Image.verify()
#
# Simple flow:
#
#     User uploads file
#            |
#            v
#          Pillow
#            |
#       +----+----+
#       |         |
#     Valid     Invalid
#       |         |
#       v         v
#      Save     Reject
#
# In simple words:
# Pillow = Image processing and image validation library.
pillow==12.3.0

# Provides the IANA timezone database used by Python's zoneinfo
# functionality.
#
# Examples of timezone names:

#     UTC

#     Europe/London
#     America/New_York
#
# The project uses timezone-aware datetime handling through
# Django's:
#
#     USE_TZ = True
#
# This is important for things such as:
#
#     - Appointment times
#     - OTP expiry times
#     - Session timestamps
#     - Created/updated timestamps
#
# On Windows, tzdata can provide timezone information that Python
# may otherwise not be able to obtain from the operating system.
#
# In simple words:
# tzdata = Timezone information for correct date/time handling.

# application on a production server.
#

#

#       |
#       v
#     Gunicorn

#       v
#     Django
#       |
#       v
#     MySQL / PostgreSQL
#
# Example command:
#
#     gunicorn config.wsgi:application
#
# Multiple workers can be used:
#
#     gunicorn --workers 3 config.wsgi:application
#
# In simple words:
# Gunicorn = Production server that runs the Django application.
